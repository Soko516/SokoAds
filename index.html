const express = require('express'); const fs = require('fs'); const path = require('path'); const crypto = require('crypto'); require('dotenv').config(); const app = express(); const PORT = Number(process.env.PORT || 3000); const PUBLIC_BASE_URL = (process.env.PUBLIC_BASE_URL || `http://localhost:${PORT}`).replace(/\/$/, ''); const ADMIN_PASSWORD = process.env.ADMIN_PASSWORD || 'admin123'; const PESAPAL_ENV = (process.env.PESAPAL_ENV || 'SANDBOX').toUpperCase(); const PESAPAL_CONSUMER_KEY = process.env.PESAPAL_CONSUMER_KEY || ''; const PESAPAL_CONSUMER_SECRET = process.env.PESAPAL_CONSUMER_SECRET || ''; const PESAPAL_IPN_ID = process.env.PESAPAL_IPN_ID || ''; const DATA_DIR = path.join(__dirname, 'data'); const ADS_FILE = path.join(DATA_DIR, 'ads.json'); const adminTokens = new Map(); const PRICING = { Basic: 2000, Featured: 5000, Premium: 10000, }; function ensureDataFiles() { if (!fs.existsSync(DATA_DIR)) fs.mkdirSync(DATA_DIR, { recursive: true }); if (!fs.existsSync(ADS_FILE)) fs.writeFileSync(ADS_FILE, '[]\n', 'utf8'); } function readAds() { ensureDataFiles(); try { const raw = fs.readFileSync(ADS_FILE, 'utf8'); const parsed = JSON.parse(raw || '[]'); return Array.isArray(parsed) ? parsed : []; } catch (error) { return []; } } function writeAds(ads) { ensureDataFiles(); fs.writeFileSync(ADS_FILE, JSON.stringify(ads, null, 2) + '\n', 'utf8'); } function normalizePackage(value) { const name = String(value || 'Basic').trim(); if (!name) return 'Basic'; return name.charAt(0).toUpperCase() + name.slice(1).toLowerCase(); } function getPriceForPackage(value) { const type = normalizePackage(value); return Number(PRICING[type] || PRICING.Basic || 0); } function slugify(value) { return String(value || '') .toLowerCase() .replace(/[^a-z0-9]+/g, '-') .replace(/^-+|-+$/g, '') .slice(0, 80) || 'ad'; } function buildPublicUrl(pathname = '/') { return `${PUBLIC_BASE_URL}${pathname.startsWith('/') ? pathname : `/${pathname}`}`; } function getPesapalBase() { if (PESAPAL_ENV === 'LIVE') return 'https://pay.pesapal.com/v3/api'; return 'https://cybqa.pesapal.com/pesapalv3/api'; } async function pesapalRequest(endpoint, options = {}) { const base = getPesapalBase(); const url = `${base}${endpoint}`; const headers = { Accept: 'application/json', ...(options.body ? { 'Content-Type': 'application/json' } : {}), ...(options.headers || {}), }; if (options.token) headers.Authorization = `Bearer ${options.token}`; const response = await fetch(url, { method: options.method || 'GET', headers, body: options.body ? JSON.stringify(options.body) : undefined, }); const text = await response.text(); let payload = {}; if (text) { try { payload = JSON.parse(text); } catch (error) { payload = { raw: text }; } } if (!response.ok) { const message = payload?.message || payload?.error || payload?.error_description || `PesaPal request failed (${response.status})`; throw new Error(message); } return payload; } async function getPesapalToken() { if (!PESAPAL_CONSUMER_KEY || !PESAPAL_CONSUMER_SECRET) { throw new Error('Missing PESAPAL_CONSUMER_KEY or PESAPAL_CONSUMER_SECRET'); } const response = await pesapalRequest('/Auth/RequestToken', { method: 'POST', body: { consumer_key: PESAPAL_CONSUMER_KEY, consumer_secret: PESAPAL_CONSUMER_SECRET, }, }); return response?.token || response?.access_token || null; } async function registerPesapalIpn() { if (!PESAPAL_CONSUMER_KEY || !PESAPAL_CONSUMER_SECRET) { return { ok: false, message: 'PesaPal credentials are not configured.' }; } const ipnUrl = buildPublicUrl('/api/payments/ipn'); const token = await getPesapalToken(); const response = await pesapalRequest('/URLSetup/RegisterIPN', { method: 'POST', token, body: { url: ipnUrl }, }); const ipnId = response?.ipn_id || response?.id || response?.notification_id || ''; return { ok: true, ipnId, url: ipnUrl, response }; } async function submitPesapalOrder({ ad, firstName, lastName, email, phone, packageType }) { const token = await getPesapalToken(); const amount = Number(ad.price || getPriceForPackage(packageType)); const orderId = `sokoads-${Date.now()}-${crypto.randomUUID().slice(0, 8)}`; const payload = { id: orderId, currency: 'TZS', amount, description: `Payment for ${ad.name || 'SokoAds listing'}`, callback_url: buildPublicUrl('/api/payments/callback'), notification_id: PESAPAL_IPN_ID, billing_address: { email_address: email, phone_number: phone, first_name: firstName, last_name: lastName, line_1: 'Dar es Salaam', city: 'Dar es Salaam', country_code: 'TZ', }, }; const response = await pesapalRequest('/Transactions/SubmitOrderRequest', { method: 'POST', token, body: payload, }); return { orderId, orderTrackingId: response?.order_tracking_id || response?.orderTrackingId || null, redirectUrl: response?.redirect_url || response?.redirectUrl || response?.payment_url || response?.checkout_url || null, amount, raw: response, }; } async function getTransactionStatus(orderTrackingId) { const token = await getPesapalToken(); return pesapalRequest(`/Transactions/GetTransactionStatus?orderTrackingId=${encodeURIComponent(orderTrackingId)}`, { method: 'GET', token, }); } function findAdById(id) { return readAds().find((ad) => ad.id === id); } function updateAdPaymentStatus(orderTrackingId, nextStatus) { const ads = readAds(); let updated = false; for (const item of ads) { if (item.payment && item.payment.orderTrackingId === orderTrackingId) { item.payment.status = nextStatus; item.status = nextStatus === 'PAID' ? 'paid' : nextStatus === 'PENDING' ? 'awaiting_payment' : 'failed'; item.updatedAt = new Date().toISOString(); updated = true; } } if (updated) writeAds(ads); return updated; } app.use(express.json({ limit: '2mb' })); app.use(express.urlencoded({ extended: true })); app.use(express.static(path.join(__dirname))); app.get('/health', (req, res) => { res.json({ ok: true, service: 'SokoAds', environment: PESAPAL_ENV, publicBaseUrl: PUBLIC_BASE_URL, hasPesapalCredentials: Boolean(PESAPAL_CONSUMER_KEY && PESAPAL_CONSUMER_SECRET), hasIpnId: Boolean(PESAPAL_IPN_ID), }); }); app.get('/api/config', (req, res) => { res.json({ environment: PESAPAL_ENV, publicBaseUrl: PUBLIC_BASE_URL, hasPesapalCredentials: Boolean(PESAPAL_CONSUMER_KEY && PESAPAL_CONSUMER_SECRET), hasIpnId: Boolean(PESAPAL_IPN_ID), }); }); app.get('/api/ads', (req, res) => { const ads = readAds().sort((a, b) => new Date(b.createdAt) - new Date(a.createdAt)); res.json(ads); }); app.post('/api/ads', (req, res) => { const payload = req.body || {}; const name = String(payload.name || '').trim(); const description = String(payload.description || '').trim(); const category = String(payload.category || 'General').trim(); const price = Number(payload.price || 0); const image = String(payload.image || '').trim(); const packageType = normalizePackage(payload.packageType || 'Basic'); if (!name || !Number.isFinite(price) || price <= 0) { return res.status(400).json({ error: 'Valid ad name and price are required.' }); } const ad = { id: crypto.randomUUID(), slug: slugify(name), name, description: description || 'New listing awaiting payment.', category, price, image: image || 'https://images.unsplash.com/photo-1523275335684-37898b6baf30?w=1200', packageType, status: 'pending', createdAt: new Date().toISOString(), updatedAt: new Date().toISOString(), payment: null, }; const ads = readAds(); ads.unshift(ad); writeAds(ads); res.status(201).json({ ad }); }); app.post('/api/ads/:id/pay', async (req, res) => { try { const ad = findAdById(req.params.id); if (!ad) return res.status(404).json({ error: 'Ad not found.' }); const firstName = String(req.body?.firstName || '').trim(); const lastName = String(req.body?.lastName || '').trim(); const email = String(req.body?.email || '').trim(); const phone = String(req.body?.phone || '').trim(); const packageType = normalizePackage(req.body?.packageType || ad.packageType || 'Basic'); if (!firstName || !lastName || !email || !phone) { return res.status(400).json({ error: 'First name, last name, email and phone are required.' }); } if (!PESAPAL_CONSUMER_KEY || !PESAPAL_CONSUMER_SECRET) { return res.status(503).json({ error: 'PesaPal credentials are not configured on the server.' }); } if (!PESAPAL_IPN_ID) { return res.status(400).json({ error: 'PESAPAL_IPN_ID is not configured. Register the public IPN URL in PesaPal and then set it in the .env file.', }); } const result = await submitPesapalOrder({ ad, firstName, lastName, email, phone, packageType, }); if (!result.redirectUrl) { return res.status(502).json({ error: 'PesaPal did not return a valid redirect URL.', details: result.raw, }); } const ads = readAds(); const item = ads.find((entry) => entry.id === ad.id); if (item) { item.status = 'awaiting_payment'; item.packageType = packageType; item.payment = { orderId: result.orderId, orderTrackingId: result.orderTrackingId, redirectUrl: result.redirectUrl, amount: result.amount, status: 'PENDING', createdAt: new Date().toISOString(), }; item.updatedAt = new Date().toISOString(); writeAds(ads); } res.json({ ok: true, redirectUrl: result.redirectUrl, orderId: result.orderId, orderTrackingId: result.orderTrackingId, amount: result.amount, }); } catch (error) { res.status(500).json({ error: 'Unable to initialize payment.', message: error.message, }); } }); app.get('/api/payments/callback', async (req, res) => { try { const orderTrackingId = req.query.orderTrackingId || req.query.order_tracking_id || ''; if (!orderTrackingId) return res.redirect('/payment-success.html?status=failed'); const statusResponse = await getTransactionStatus(orderTrackingId); const status = String(statusResponse?.status || statusResponse?.payment_status || statusResponse?.transaction_status || 'PENDING').toUpperCase(); updateAdPaymentStatus(orderTrackingId, status); return res.redirect(`/payment-success.html?status=${status === 'PAID' ? 'success' : 'failed'}&order=${encodeURIComponent(orderTrackingId)}`); } catch (error) { return res.redirect('/payment-success.html?status=failed'); } }); app.post('/api/payments/callback', async (req, res) => { try { const payload = req.body || {}; const orderTrackingId = payload.orderTrackingId || payload.order_tracking_id || ''; if (!orderTrackingId) { return res.status(400).json({ ok: false, message: 'Missing orderTrackingId' }); } const statusResponse = await getTransactionStatus(orderTrackingId); const status = String(statusResponse?.status || statusResponse?.payment_status || statusResponse?.transaction_status || 'PENDING').toUpperCase(); updateAdPaymentStatus(orderTrackingId, status); res.json({ ok: true, status }); } catch (error) { res.status(500).json({ ok: false, message: error.message || 'Callback failed' }); } }); app.post('/api/payments/ipn', async (req, res) => { try { const payload = req.body || {}; const orderTrackingId = payload.orderTrackingId || payload.order_tracking_id || payload.OrderTrackingId || ''; const status = String(payload.status || payload.transaction_status || 'PENDING').toUpperCase(); if (orderTrackingId) { updateAdPaymentStatus(orderTrackingId, status); } res.json({ ok: true, received: true, orderTrackingId, status }); } catch (error) { res.status(500).json({ ok: false, message: 'IPN handler failed' }); } }); app.post('/api/admin/login', (req, res) => { const password = String(req.body?.password || ''); if (!password || password !== ADMIN_PASSWORD) { return res.status(401).json({ error: 'Invalid admin password.' }); } const token = crypto.randomUUID(); adminTokens.set(token, { createdAt: Date.now() }); return res.json({ ok: true, token }); }); app.get('/api/admin/ads', (req, res) => { const token = String(req.headers['x-admin-token'] || ''); if (!adminTokens.has(token)) { return res.status(401).json({ error: 'Admin token required.' }); } return res.json(readAds()); }); app.post('/api/admin/register-ipn', async (req, res) => { try { const token = String(req.headers['x-admin-token'] || ''); if (!adminTokens.has(token)) { return res.status(401).json({ error: 'Admin token required.' }); } const result = await registerPesapalIpn(); if (!result.ok) { return res.status(400).json(result); } res.json(result); } catch (error) { res.status(500).json({ ok: false, error: error.message }); } }); app.get('*', (req, res) => { const indexFile = path.join(__dirname, 'index.html'); if (fs.existsSync(indexFile)) return res.sendFile(indexFile); res.status(404).send('Not found'); }); app.listen(PORT, () => { console.log(`SokoAds running on http://localhost:${PORT}`); console.log(`PesaPal environment: ${PESAPAL_ENV}`); console.log(`Has PesaPal credentials: ${Boolean(PESAPAL_CONSUMER_KEY && PESAPAL_CONSUMER_SECRET)}`); console.log(`Has IPN ID: ${Boolean(PESAPAL_IPN_ID)}`); });
